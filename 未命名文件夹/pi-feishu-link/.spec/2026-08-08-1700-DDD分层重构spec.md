## 2. 目标架构（DDD 四层）

```
src/
├── index.ts                # 薄装配层（<800 行）：构建 BridgeContext + 注册事件/命令/生命周期
├── application/            # 应用服务层（编排，依赖倒置——通过 BridgeContext 注入）
│   ├── bridge-context.ts      # BridgeContext 接口（所有编排依赖的装配对象）
│   ├── command-router.ts      # 命令分发（handleCommand：桥特有/pi 内置/转发）
│   ├── message-handler.ts     # 消息编排（handleInbound + handleConversationMessage）
│   ├── notification-service.ts# notify*（通知）
│   ├── diagnostics-service.ts # 诊断导出+发文件
│   └── status-formatter.ts    # 状态行/明细（纯函数）
├── commands/               # 命令层（domain）
│   ├── pi-command-adapter.ts  # pi 原生命令适配（已有）
│   └── command-controller.ts  # ⚠ 合并：classifyCommand 废弃，仅保留桥特有命令常量
├── inbound/                # 飞书接入（domain，已有）
├── sessions/               # pi 编排（domain，已有）
├── outbound/               # 可靠投递（domain，已有）
├── host/                   # 基础设施（已有）
├── common/                 # 共享（已有）
└── presentation/           # 渲染（已有）
```

**依赖方向**：application → domain（inbound/sessions/outbound）+ presentation + common；index.ts 只装配。

## 4. BridgeContext 设计（依赖倒置）

```ts
// src/application/bridge-context.ts
export interface BridgeContext {
  // 域服务
  conversations: ConversationManager | undefined;
  transport: FeishuTransport | undefined;
  outbox: Outbox | undefined;
  router: OutboundRouter;
  statusStore: StatusStore;
  liveChannel: LiveChannel | undefined;
  eventForwarder: EventForwarder | undefined;
  bridgeRuntime: BridgeRuntime | undefined;
  permissionBridge: PermissionBridge | undefined;
  turnSupervisor: TurnSupervisor | undefined;
  piBackend: PiSessionBackend | undefined;
  // 基础设施
  logger: Logger;
  cfg: () => FeishuConfig | undefined;
  // 可变状态
  botOpenId?: string;
  started: boolean;
  streamCards: Map<string, { messageId: string; text: string; touchedAt: number }>;
  streamTargets: Map<string, { messageId?: string; chatId: string }>;
  gatewayLock?: { owner: GatewayOwner; release(): Promise<void>; update(s: string): void };
  // 工具
  notifyOwner(text: string): Promise<void>;
  notifyConversation(key: string, text: string): Promise<void>;
  replyTo(msg: FeishuInboundMessage, textOrCard: string | unknown): Promise<void>;
  conversationKeyFor(msg: FeishuInboundMessage): string;
  markDone(msg: FeishuInboundMessage): void;
}
```

**依赖注入**：各 service 接收 ctx（或最小依赖集），index.ts 装配 ctx 后调用。
